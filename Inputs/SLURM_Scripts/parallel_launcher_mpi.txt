#!/usr/bin/env bash

set -euo pipefail


# ============================================================
# Configuration
# ============================================================

PYTHON_SCRIPT="/Users/haydenmonk/Downloads/HPCC/Mass_Fraction/Inputs/N_Body_Scripts/local_disc_mpi.py"
PARAM_FILE="/path/to/parameter_file"

# Number of planet masses in your Python tmax_mass_list
N_MASSES=8

# Number of MPI particles/ranks to run simultaneously PER mass.
# Keep this small for local testing.
MPI_RANKS_PER_MASS=8

# Number of different masses GNU Parallel is allowed to run
# simultaneously.
MAX_CONCURRENT_MASSES=1

# Unique ID for this local launcher run.
LOCAL_JOB_ID="local_$(date +%Y%m%d_%H%M%S)"


echo "========================================"
echo "Local MPI + GNU Parallel run"
echo "Job ID:                $LOCAL_JOB_ID"
echo "Masses:                $N_MASSES"
echo "MPI ranks per mass:    $MPI_RANKS_PER_MASS"
echo "Concurrent masses:     $MAX_CONCURRENT_MASSES"
echo "========================================"


# ============================================================
# Launch
# ============================================================

seq 0 $((N_MASSES - 1)) |
parallel \
    --jobs "$MAX_CONCURRENT_MASSES" \
    --halt soon,fail=1 \
    --line-buffer \
    '
    echo "Starting mass index {}"

    mpirun -np '"$MPI_RANKS_PER_MASS"' \
        python -u "'"$PYTHON_SCRIPT"'" \
        "'"$LOCAL_JOB_ID"'" \
        {} \
        "'"$PARAM_FILE"'"

    echo "Finished mass index {}"
    '
